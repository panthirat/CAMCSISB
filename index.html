create table if not exists public.auditorium_events ( id text primary key, event_name text not null unique, seat_data jsonb not null default '{}'::jsonb, extra_seat_list jsonb not null default '[]'::jsonb, balcony_seat_list jsonb not null default '{"L": [], "R": []}'::jsonb, walkway_seat_list jsonb not null default '[]'::jsonb, updated_at timestamptz not null default now() ); alter table public.auditorium_events enable row level security; drop policy if exists "Allow full access to auditorium_events" on public.auditorium_events; create policy "Allow full access to auditorium_events" on public.auditorium_events for all using (true) with check (true); do $$ begin if not exists (select 1 from pg_publication_tables where pubname = 'supabase_realtime' and schemaname = 'public' and tablename = 'auditorium_events') then alter publication supabase_realtime add table public.auditorium_events; end if; end $$;
